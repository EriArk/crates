cmake_minimum_required(VERSION 3.24)
project(SessionTimer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick Network)
find_package(SDL2 REQUIRED)
qt_add_executable(session-timer src/main.cpp src/Timer.h src/Timer.cpp src/Model.h)
qt_add_resources(session-timer ui PREFIX "/" FILES ui/Main.qml)
target_link_libraries(session-timer PRIVATE Qt6::Quick Qt6::Network SDL2::SDL2)
install(TARGETS session-timer RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(timer-tests tests/TimerTests.cpp src/Timer.cpp src/Timer.h src/Model.h)
    target_include_directories(timer-tests PRIVATE src)
    target_link_libraries(timer-tests PRIVATE Qt6::Quick Qt6::Network Qt6::Test)
    add_test(NAME session_timer COMMAND timer-tests)
    # Optional consumer acceptance against an already-built real TrainerOS host.
    set(TRAINEROS_SOURCE "" CACHE PATH "TrainerOS SDK source checkout")
    set(TRAINEROS_CORE "" CACHE FILEPATH "Matching built trainer_core static library")
    if(TRAINEROS_SOURCE AND TRAINEROS_CORE)
        find_package(Qt6 REQUIRED COMPONENTS WebSockets Sql Xml Multimedia Concurrent)
        find_package(Qt6Keychain CONFIG REQUIRED)
        find_package(OpenSSL REQUIRED)
        qt_add_executable(timer-host-tests tests/HostIntegration.cpp src/Timer.cpp src/Timer.h src/Model.h)
        qt_add_resources(timer-host-tests test_ui PREFIX "/" FILES ui/Main.qml)
        target_include_directories(timer-host-tests PRIVATE src "${TRAINEROS_SOURCE}/src")
        target_link_libraries(timer-host-tests PRIVATE "${TRAINEROS_CORE}" Qt6::Test Qt6::Quick
            Qt6::Network Qt6::WebSockets Qt6::Sql Qt6::Xml Qt6::Multimedia Qt6::Concurrent
            Qt6Keychain::Qt6Keychain OpenSSL::Crypto SDL2::SDL2)
        add_test(NAME session_timer_host COMMAND timer-host-tests $<TARGET_FILE:session-timer>)
    endif()
endif()
