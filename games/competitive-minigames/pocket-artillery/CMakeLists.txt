cmake_minimum_required(VERSION 3.24)
project(PocketArtillery VERSION 0.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick Network)
find_package(SDL2 REQUIRED)
add_library(artillery-game STATIC ../avatar-sdk/AvatarCards.h src/Game.h src/Game.cpp src/Bridge.h src/Bridge.cpp src/World.h)
target_include_directories(artillery-game PUBLIC src ../avatar-sdk)
target_link_libraries(artillery-game PUBLIC Qt6::Quick Qt6::Network)
set(AVATAR_UI)
foreach(name AvatarPortrait AvatarPortraitScene PlayerCard)
    set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/../avatar-sdk/${name}.qml PROPERTIES QT_RESOURCE_ALIAS ui/${name}.qml)
    list(APPEND AVATAR_UI ${CMAKE_CURRENT_SOURCE_DIR}/../avatar-sdk/${name}.qml)
endforeach()
qt_add_executable(pocket-artillery src/main.cpp)
qt_add_resources(pocket-artillery ui PREFIX "/" FILES ui/Main.qml ui/Battlefield.qml ${AVATAR_UI})
target_link_libraries(pocket-artillery PRIVATE artillery-game SDL2::SDL2)
install(TARGETS pocket-artillery RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(artillery-tests tests/GameTests.cpp)
    qt_add_resources(artillery-tests test_ui PREFIX "/" FILES ui/Main.qml ui/Battlefield.qml ${AVATAR_UI})
    target_link_libraries(artillery-tests PRIVATE artillery-game Qt6::Test SDL2::SDL2)
    add_test(NAME artillery COMMAND artillery-tests -o game-results.txt,txt -o -,txt)
    set_tests_properties(artillery PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    set(TRAINEROS_SOURCE "" CACHE PATH "TrainerOS SDK source checkout")
    set(TRAINEROS_CORE "" CACHE FILEPATH "Matching built trainer_core static library")
    if(TRAINEROS_SOURCE AND TRAINEROS_CORE)
        find_package(Qt6 REQUIRED COMPONENTS WebSockets Sql Xml Multimedia Concurrent)
        find_package(Qt6Keychain CONFIG REQUIRED)
        find_package(OpenSSL REQUIRED)
        qt_add_executable(artillery-host-tests tests/HostIntegration.cpp)
        target_include_directories(artillery-host-tests PRIVATE "${TRAINEROS_SOURCE}/src")
        target_compile_definitions(artillery-host-tests PRIVATE SDK_SOURCE="${TRAINEROS_SOURCE}")
        target_link_libraries(artillery-host-tests PRIVATE artillery-game "${TRAINEROS_CORE}" Qt6::Test Qt6::WebSockets Qt6::Sql Qt6::Xml Qt6::Multimedia Qt6::Concurrent Qt6Keychain::Qt6Keychain OpenSSL::Crypto SDL2::SDL2)
        add_test(NAME artillery_host COMMAND artillery-host-tests -o host-results.txt,txt -o -,txt)
        set_tests_properties(artillery_host PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    endif()
endif()
