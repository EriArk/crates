cmake_minimum_required(VERSION 3.24)
project(CratesArcade VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick Network)
find_package(SDL2 REQUIRED)
set(COMMON "${CMAKE_CURRENT_SOURCE_DIR}/arcade-common")
add_library(arcade STATIC
    ${COMMON}/Rules.h ${COMMON}/Rules.cpp ${COMMON}/ArcadeBridge.h ${COMMON}/ArcadeBridge.cpp
    ${COMMON}/ArcadeGame.h ${COMMON}/ArcadeGame.cpp
    pong/src/PongRules.h pong/src/PongRules.cpp
    snake-arena/src/SnakeRules.h snake-arena/src/SnakeRules.cpp
    light-cycles/src/CyclesRules.h light-cycles/src/CyclesRules.cpp
    ${COMMON}/PuzzleRules.h
    falling-blocks/src/BlocksRules.h falling-blocks/src/BlocksRules.cpp
    minesweeper-race/src/MinesRules.h minesweeper-race/src/MinesRules.cpp
    chain-reaction/src/ChainRules.h chain-reaction/src/ChainRules.cpp
    mini-racing/src/RacingRules.h mini-racing/src/RacingRules.cpp
    tanks/src/TanksRules.h tanks/src/TanksRules.cpp)
target_include_directories(arcade PUBLIC ${COMMON} ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(arcade PUBLIC Qt6::Quick Qt6::Network)
set_source_files_properties(${COMMON}/ui/Main.qml PROPERTIES QT_RESOURCE_ALIAS ui/Main.qml)
set_source_files_properties(${COMMON}/ui/ImpactFx.qml PROPERTIES QT_RESOURCE_ALIAS ui/ImpactFx.qml)
set(ARCADE_RESOURCES ${COMMON}/ui/Main.qml ${COMMON}/ui/ImpactFx.qml)
foreach(slug pong snake-arena light-cycles falling-blocks minesweeper-race chain-reaction mini-racing tanks)
    set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/${slug}/ui/Board.qml PROPERTIES QT_RESOURCE_ALIAS boards/${slug}.qml)
    list(APPEND ARCADE_RESOURCES ${CMAKE_CURRENT_SOURCE_DIR}/${slug}/ui/Board.qml)
endforeach()
set(ARCADE_GAMES "pong;snake-arena;light-cycles;falling-blocks;minesweeper-race;chain-reaction;mini-racing;tanks" CACHE STRING "Games to build")
foreach(slug IN LISTS ARCADE_GAMES)
    if(slug STREQUAL "pong")
        set(kind 0)
    elseif(slug STREQUAL "snake-arena")
        set(kind 1)
    elseif(slug STREQUAL "light-cycles")
        set(kind 2)
    elseif(slug STREQUAL "falling-blocks")
        set(kind 3)
    elseif(slug STREQUAL "minesweeper-race")
        set(kind 4)
    elseif(slug STREQUAL "chain-reaction")
        set(kind 5)
    elseif(slug STREQUAL "mini-racing")
        set(kind 6)
    elseif(slug STREQUAL "tanks")
        set(kind 7)
    else()
        message(FATAL_ERROR "Unknown arcade game: ${slug}")
    endif()
    qt_add_executable(${slug} ${COMMON}/main.cpp)
    target_compile_definitions(${slug} PRIVATE ARCADE_KIND=${kind})
    target_link_libraries(${slug} PRIVATE arcade SDL2::SDL2)
    qt_add_resources(${slug} ${slug}_ui PREFIX "/" FILES ${ARCADE_RESOURCES})
    install(TARGETS ${slug} RUNTIME DESTINATION bin)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(arcade-tests ${COMMON}/tests/ArcadeTests.cpp)
    target_link_libraries(arcade-tests PRIVATE arcade Qt6::Test SDL2::SDL2)
    qt_add_resources(arcade-tests test_ui PREFIX "/" FILES ${ARCADE_RESOURCES})
    add_test(NAME arcade COMMAND arcade-tests -o arcade-results.txt,txt -o -,txt)
    set_tests_properties(arcade PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    set(TRAINEROS_SOURCE "" CACHE PATH "Matching TrainerOS SDK checkout")
    set(TRAINEROS_CORE "" CACHE FILEPATH "Matching trainer_core static library")
    if(TRAINEROS_SOURCE AND TRAINEROS_CORE)
        find_package(Qt6 REQUIRED COMPONENTS WebSockets Sql Xml Multimedia Concurrent)
        find_package(Qt6Keychain CONFIG REQUIRED)
        find_package(OpenSSL REQUIRED)
        qt_add_executable(arcade-host-tests ${COMMON}/tests/HostTests.cpp)
        target_include_directories(arcade-host-tests PRIVATE ${TRAINEROS_SOURCE}/src)
        target_compile_definitions(arcade-host-tests PRIVATE SDK_SOURCE="${TRAINEROS_SOURCE}")
        target_link_libraries(arcade-host-tests PRIVATE arcade ${TRAINEROS_CORE} Qt6::Test Qt6::WebSockets Qt6::Sql Qt6::Xml Qt6::Multimedia Qt6::Concurrent Qt6Keychain::Qt6Keychain OpenSSL::Crypto SDL2::SDL2)
        add_test(NAME arcade_host COMMAND arcade-host-tests -o host-results.txt,txt -o -,txt)
    endif()
endif()
