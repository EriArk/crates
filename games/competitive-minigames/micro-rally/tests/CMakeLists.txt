cmake_minimum_required(VERSION 3.24)
project(RallyHostTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Network Test WebSockets Sql Xml Multimedia Concurrent Widgets)
find_package(Qt6Keychain CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SDL2 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_library(FM_QT_LIBRARY NAMES fm-qt6 REQUIRED)
set(TRAINEROS_SOURCE "" CACHE PATH "Matching SDK source")
set(TRAINEROS_CORE "" CACHE FILEPATH "Matching trainer_core library")
qt_add_executable(rally-host-tests HostIntegration.cpp)
target_include_directories(rally-host-tests PRIVATE "${TRAINEROS_SOURCE}/src")
target_compile_definitions(rally-host-tests PRIVATE SDK_SOURCE="${TRAINEROS_SOURCE}")
target_link_libraries(rally-host-tests PRIVATE "${TRAINEROS_CORE}" Qt6::Quick Qt6::Network Qt6::Test Qt6::WebSockets Qt6::Sql Qt6::Xml Qt6::Multimedia Qt6::Concurrent Qt6Keychain::Qt6Keychain OpenSSL::Crypto SDL2::SDL2 PkgConfig::GIO ${FM_QT_LIBRARY} Qt6::Widgets)
