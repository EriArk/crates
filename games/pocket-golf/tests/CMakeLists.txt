cmake_minimum_required(VERSION 3.24)
project(GolfHostTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick Network Test WebSockets Sql Xml Multimedia Concurrent)
find_package(Qt6Keychain CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SDL2 REQUIRED)
option(TRAINEROS_CORE_WITH_LIBFM "Match a Linux trainer_core built with libfm" OFF)
set(TRAINEROS_SOURCE "" CACHE PATH "Matching SDK source")
set(TRAINEROS_CORE "" CACHE FILEPATH "Matching trainer_core library")
qt_add_executable(golf-host-tests HostIntegration.cpp)
target_include_directories(golf-host-tests PRIVATE "${TRAINEROS_SOURCE}/src")
target_compile_definitions(golf-host-tests PRIVATE SDK_SOURCE="${TRAINEROS_SOURCE}")
target_link_libraries(golf-host-tests PRIVATE "${TRAINEROS_CORE}" Qt6::Quick Qt6::Network Qt6::Test Qt6::WebSockets Qt6::Sql Qt6::Xml Qt6::Multimedia Qt6::Concurrent Qt6Keychain::Qt6Keychain OpenSSL::Crypto SDL2::SDL2)
if(TRAINEROS_CORE_WITH_LIBFM)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
    find_library(FM_QT_LIBRARY NAMES fm-qt6 REQUIRED)
    target_link_libraries(golf-host-tests PRIVATE PkgConfig::GIO ${FM_QT_LIBRARY} Qt6::Widgets)
endif()
